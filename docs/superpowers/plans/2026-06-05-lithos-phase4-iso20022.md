# Lithos Phase 4 — ISO 20022 Settlement Messaging + Token-Settlement Signal Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let Lithos generate a `pain.001` commission payment-instruction, ingest a `camt.054` bank credit notification to auto-confirm the commission, and anchor a non-custodial on-chain settlement signal — all backend-only.

**Architecture:** A new `backend/src/iso20022/` module with *pure* XML builder/parser functions plus a thin orchestration service and controller. The on-chain settlement signal is one new `AnchorType` value and a `signalSettlement()` method on the existing `BlockchainAnchorService` (so it stays inside the non-custodial guard). No funds ever move; Lithos produces and reads documents.

**Tech Stack:** NestJS · Prisma (PostgreSQL) · Jest · hand-rolled XML (no new dependency).

**Working directory for all commands:** `normie-apps/lithium-broker/backend`

**Non-custodial guard:** `test/non-custodial.spec.ts` must stay green — no `transferFrom`/`releaseFunds`/`escrowRelease`/`payout`/`custody` anywhere in `src`, no `escrow` module, `sendTransaction` only in `src/blockchain/`. This plan introduces none of those.

**Spec:** `docs/superpowers/specs/2026-06-05-lithos-phase4-iso20022-design.md`

```ts
export interface Pain001Party {
  name: string;
  iban?: string;
  bic?: string;
}

export interface Pain001Input {
  messageId: string;
  creationDateTime: string; // ISO 8601 — injected by the caller (deterministic)
  debtor: Pain001Party; // the paying party named on the deal
  creditor: Pain001Party; // the broker/mandate receiving the commission
  amount: string; // decimal string, e.g. "12500.00"
  currency: string; // ISO 4217, e.g. "USD"
  endToEndId: string; // the deal number
  remittanceInfo: string;
}

function escapeXml(value: string): string {
  return value
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&apos;");
}

/**
 * Build a pain.001.001.09 CustomerCreditTransferInitiation message.
 * Pure: no I/O, no Date, no funds — just a payment INSTRUCTION document the
 * broker submits to their own bank. Account details Lithos does not hold are
 * emitted as documented placeholders the broker completes before submission.
 */
export function buildPain001(input: Pain001Input): string {
  const dbtrAcct = input.debtor.iban ?? "IBAN-TO-COMPLETE";
  const cdtrAcct = input.creditor.iban ?? "IBAN-TO-COMPLETE";
  const cdtrBic = input.creditor.bic ?? "BIC-TO-COMPLETE";
  return `<?xml version="1.0" encoding="UTF-8"?>
<Document xmlns="urn:iso:std:iso:20022:tech:xsd:pain.001.001.09">
  <CstmrCdtTrfInitn>
    <GrpHdr>
      <MsgId>${escapeXml(input.messageId)}</MsgId>
      <CreDtTm>${escapeXml(input.creationDateTime)}</CreDtTm>
      <NbOfTxs>1</NbOfTxs>
      <CtrlSum>${input.amount}</CtrlSum>
      <InitgPty><Nm>${escapeXml(input.debtor.name)}</Nm></InitgPty>
    </GrpHdr>
    <PmtInf>
      <PmtInfId>${escapeXml(input.messageId)}-PMT</PmtInfId>
      <PmtMtd>TRF</PmtMtd>
      <NbOfTxs>1</NbOfTxs>
      <CtrlSum>${input.amount}</CtrlSum>
      <Dbtr><Nm>${escapeXml(input.debtor.name)}</Nm></Dbtr>
      <DbtrAcct><Id><IBAN>${escapeXml(dbtrAcct)}</IBAN></Id></DbtrAcct>
      <CdtTrfTxInf>
        <PmtId><EndToEndId>${escapeXml(input.endToEndId)}</EndToEndId></PmtId>
        <Amt><InstdAmt Ccy="${escapeXml(input.currency)}">${input.amount}</InstdAmt></Amt>
        <CdtrAgt><FinInstnId><BICFI>${escapeXml(cdtrBic)}</BICFI></FinInstnId></CdtrAgt>
        <Cdtr><Nm>${escapeXml(input.creditor.name)}</Nm></Cdtr>
        <CdtrAcct><Id><IBAN>${escapeXml(cdtrAcct)}</IBAN></Id></CdtrAcct>
        <RmtInf><Ustrd>${escapeXml(input.remittanceInfo)}</Ustrd></RmtInf>
      </CdtTrfTxInf>
    </PmtInf>
  </CstmrCdtTrfInitn>
</Document>`;
}
```

```ts
import { BadRequestException } from "@nestjs/common";

export interface Camt054Notification {
  amount: string;
  currency: string;
  creditDebitIndicator: "CRDT" | "DBIT";
  endToEndId?: string;
  debtorName?: string;
}

/**
 * Parse the first entry of a camt.054.001.08 BankToCustomerDebitCreditNotification.
 * Scope-limited to the fields Lithos needs to match a commission credit.
 * XXE-safe: any DOCTYPE/ENTITY declaration is rejected outright (a legitimate
 * camt.054 never carries one), so no external entity is ever resolved.
 */
export function parseCamt054(xml: string): Camt054Notification {
  if (/<!DOCTYPE/i.test(xml) || /<!ENTITY/i.test(xml)) {
    throw new BadRequestException("camt.054: DOCTYPE/ENTITY declarations are not allowed");
  }

  const amtMatch = xml.match(/<Amt[^>]*\bCcy="([^"]+)"[^>]*>([^<]+)<\/Amt>/);
  const indMatch = xml.match(/<CdtDbtInd>\s*(CRDT|DBIT)\s*<\/CdtDbtInd>/);
  if (!amtMatch || !indMatch) {
    throw new BadRequestException("camt.054: could not parse amount and credit/debit indicator");
  }

  const e2eMatch = xml.match(/<EndToEndId>([^<]+)<\/EndToEndId>/);
  const dbtrMatch = xml.match(/<Dbtr>\s*<Nm>([^<]+)<\/Nm>/);

  return {
    currency: amtMatch[1].trim(),
    amount: amtMatch[2].trim(),
    creditDebitIndicator: indMatch[1] as "CRDT" | "DBIT",
    endToEndId: e2eMatch ? e2eMatch[1].trim() : undefined,
    debtorName: dbtrMatch ? dbtrMatch[1].trim() : undefined,
  };
}
```

```ts
  /**
   * Non-custodial settlement signal: fingerprint the deal's final settlement
   * state and anchor it on testnet (reuses the simulated-tx stub — no funds,
   * no real chain tx). A tamper-evident milestone record, not a value transfer.
   */
  async signalSettlement(dealId: string, actorId: string) {
    const deal = await this.prisma.deal.findUnique({ where: { id: dealId } });
    if (!deal) throw new NotFoundException("Deal not found");

    const fingerprint = this.createFingerprint(
      JSON.stringify({
        dealNumber: deal.dealNumber,
        volume: deal.volume,
        currency: deal.currency,
        commissionValue: deal.commissionValue,
        commissionBasis: deal.commissionBasis,
        status: deal.status,
      }),
    );

    return this.anchor(
      {
        dealId,
        entityType: AnchorType.SETTLEMENT_SIGNAL,
        fingerprint,
        metadata: { kind: "settlement" },
      },
      actorId,
    );
  }
```

```ts
  constructor(
    private readonly prisma: PrismaService,
    private readonly events: DealEventsService,
    private readonly anchor: BlockchainAnchorService,
  ) {}

  /** Derive the broker commission amount from the deal. Throws if indeterminable. */
  computeCommission(deal: Deal): { amount: string; currency: string } {
    const currency = deal.currency ?? "USD";
    const basis = deal.commissionBasis;
    const value = deal.commissionValue != null ? Number(deal.commissionValue) : null;
    const volume = deal.volume != null ? Number(deal.volume) : null;
    if (!basis || value == null) {
      throw new BadRequestException("Commission amount is not determinable");
    }

    let amount: number;
    if (basis === "FLAT") {
      amount = value;
    } else if (basis === "PER_TONNE") {
      if (volume == null) throw new BadRequestException("Commission amount is not determinable");
      amount = value * volume;
    } else if (basis === "PERCENT") {
      const price = deal.manualPrice != null ? Number(deal.manualPrice)
        : deal.priceFloor != null ? Number(deal.priceFloor) : null;
      if (price == null || volume == null) {
        throw new BadRequestException("Commission amount is not determinable");
      }
      amount = price * volume * (value / 100);
    } else {
      // SPREAD and any future basis are not supported this pass.
      throw new BadRequestException("Commission amount is not determinable");
    }
    return { amount: amount.toFixed(2), currency };
  }

  private async loadSettledDeal(dealId: string): Promise<Deal> {
    const deal = await this.prisma.deal.findUnique({ where: { id: dealId } });
    if (!deal) throw new NotFoundException("Deal not found");
    if (deal.phase !== LithiumPhase.SETTLEMENT) {
      throw new BadRequestException("Settlement messaging is only available at SETTLEMENT");
    }
    return deal;
  }

  /** Generate a pain.001 commission payment instruction and persist it as a document. */
  async generatePain001(dealId: string, actorId: string, creationDateTime?: string) {
    const deal = await this.loadSettledDeal(dealId);
    const { amount, currency } = this.computeCommission(deal);

    const xml = buildPain001({
      messageId: `LITHOS-${deal.dealNumber}`,
      creationDateTime: creationDateTime ?? new Date().toISOString(),
      debtor: { name: "Deal paying party (complete account before submission)" },
      creditor: { name: "Lithos brokerage (mandate)" },
      amount,
      currency,
      endToEndId: deal.dealNumber,
      remittanceInfo: `Commission for deal ${deal.dealNumber}`,
    });

    const doc = await this.prisma.document.create({
      data: {
        dealId,
        type: DocumentType.PAIN001,
        content: xml,
        documentUrl: `generated://pain001/${dealId}`,
        uploadedBy: actorId,
      },
    });

    await this.events.record({
      dealId,
      eventType: DealEventType.PAYMENT_INSTRUCTION_GENERATED,
      actorRole: "ADMIN",
      actorId,
      metadata: { documentId: doc.id, amount, currency },
    });

    return { documentId: doc.id, xml };
  }

  /** Ingest a camt.054 credit notification; on a match, confirm the commission. */
  async ingestCamt054(dealId: string, xml: string, actorId: string) {
    const deal = await this.loadSettledDeal(dealId);
    const parsed = parseCamt054(xml);

    const doc = await this.prisma.document.create({
      data: {
        dealId,
        type: DocumentType.CAMT054,
        content: xml,
        documentUrl: `ingested://camt054/${dealId}`,
        uploadedBy: actorId,
      },
    });

    const { amount: expected, currency } = this.computeCommission(deal);
    const matched =
      parsed.creditDebitIndicator === "CRDT" &&
      parsed.currency === currency &&
      Math.abs(Number(parsed.amount) - Number(expected)) <= AMOUNT_TOLERANCE;

    if (matched) {
      await this.prisma.deal.update({
        where: { id: dealId },
        data: { commissionStatus: CommissionStatus.RECORDED_PAID },
      });
      await this.events.record({
        dealId,
        eventType: DealEventType.COMMISSION_RECORDED,
        actorRole: "ADMIN",
        actorId,
        metadata: { source: "CAMT054", amount: parsed.amount, currency, documentId: doc.id },
      });
      await this.anchor.signalSettlement(dealId, actorId);
    } else {
      await this.events.record({
        dealId,
        eventType: DealEventType.SETTLEMENT_MISMATCH,
        actorRole: "ADMIN",
        actorId,
        metadata: {
          expected,
          received: parsed.amount,
          expectedCurrency: currency,
          receivedCurrency: parsed.currency,
          indicator: parsed.creditDebitIndicator,
          documentId: doc.id,
        },
      });
    }

    return { matched, expected, received: parsed.amount, documentId: doc.id };
  }
}
```
